name: Build ImmortalWrt

on:
  push:
    tags:
      - v**
  workflow_dispatch:

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest

    container:
      image: ubuntu:20.04

    strategy:
      matrix:
        include:
          - config_name: minimal
            config_file: minimal
          - config_name: full
            config_file: full

    env:
      DEBIAN_FRONTEND: noninteractive
      TZ: Europe/Paris

    steps:
      - name: Checkout My Repository Source
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Initialize Build Environment
        run: |
          apt-get update
          apt-get install -y \
            ca-certificates \
            curl \
            tzdata

          ln -snf /usr/share/zoneinfo/Europe/Paris /etc/localtime
          echo "Europe/Paris" > /etc/timezone

          bash <(curl -fsSL https://build-scripts.immortalwrt.org/init_build_environment.sh)

      - name: Update and Install Feeds
        run: |
          ./scripts/feeds update -a
          ./scripts/feeds install -a

      - name: Load Configuration (${{ matrix.config_name }})
        run: |
          CONFIG_FILE="defconfig/${{ matrix.config_file }}.config"

          if [ ! -f "$CONFIG_FILE" ]; then
            echo "Config file not found: $CONFIG_FILE"
            exit 1
          fi

          cp -f "$CONFIG_FILE" .config
          echo "Successfully loaded $CONFIG_FILE"

          make defconfig

          echo "Key config:"
          grep -E '^(CONFIG_TARGET_|CONFIG_PACKAGE_luci-app-passwall=|CONFIG_PACKAGE_luci-app-ttyd=|CONFIG_PACKAGE_ttyd=)' .config || true

      - name: Download Sources
        run: |
          make download -j"$(nproc)"
          find dl -type f -size -1024c -delete

      - name: Compile Firmware
        run: |
          set -euo pipefail

          if ! make -j"$(nproc)"; then
            echo "::warning::Parallel build failed, retrying with -j1 V=s"
            make -j1 V=s
          fi

      - name: Upload Artifacts
        uses: actions/upload-artifact@v4
        with:
          name: firmware-${{ matrix.config_name }}
          path: bin/targets/mediatek/filogic/*.bin

  publish:
    needs: build
    runs-on: ubuntu-latest
    permissions:
      contents: write
    if: ${{ github.event_name == 'push' && github.ref_type == 'tag' }}

    steps:
      - name: Download All Artifacts
        uses: actions/download-artifact@v4
        with:
          path: all-firmwares

      - name: Flatten and Rename Files
        run: |
          mkdir -p final-release

          for config_dir in all-firmwares/*/; do
            config_name=$(basename "$config_dir")

            for bin_file in "$config_dir"*.bin; do
              if [ -f "$bin_file" ]; then
                base=$(basename "$bin_file")
                cp "$bin_file" "final-release/${base%.bin}-${config_name}.bin"
              fi
            done
          done

      - name: Create Unified Release
        uses: softprops/action-gh-release@v2
        with:
          make_latest: true
          tag_name: ${{ github.ref_name }}
          files: final-release/*.bin
          draft: true
          body: |
            ## 📦 发布说明

            ### ✨ 构建配置
            - **minimal**: 最小化构建，仅包含核心功能。
            - **full**: 完整构建，包含以下额外预装插件：
              - `luci-proto-wireguard`: WireGuard 协议支持
              - `luci-app-wol`: 网络唤醒 (WoL) 功能
              - `luci-i18n-wol-zh-cn`: WoL 中文语言包
              - `ddns-scripts-cloudflare`: Cloudflare DDNS 脚本
              - `luci-app-ddns`: 动态 DNS 配置界面
              - `luci-i18n-ddns-zh-cn`: DDNS 中文语言包
              - `bind-host`: BIND DNS 服务器宿主支持
              - `luci-app-store`: iStore 软件中心
              - `kmod-tun` / `kmod-iptunnel` / `kmod-wireguard`: VPN 隧道基础模块
              - `luci-app-ttyd` / `ttyd`: 网页终端
            - **passwall**: 单独 workflow 构建，基于 full 额外包含 Passwall，并追加到同一发布。
